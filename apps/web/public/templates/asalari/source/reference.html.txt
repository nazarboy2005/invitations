  try { return JSON.parse(localStorage.getItem(WK) || '[]'); } catch { return []; }
}
function saveWishes(list){ localStorage.setItem(WK, JSON.stringify(list)); }

function renderFeed(){
  const list = loadWishes();
  if (!list.length){
    feed.innerHTML = '<div class="empty">be the first to leave a wish ✿</div>';
    return;
  }
  feed.innerHTML = list.slice().reverse().map(w=>{
    const t = new Date(w.t);
    const when = t.toLocaleDateString(undefined, {month:'short', day:'numeric'}) + ' · ' +
                 t.toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'});
    return `<div class="wish">
      <div class="who">${escapeHTML(w.name)} <span class="when">· ${when}</span></div>
      <div class="msg">${escapeHTML(w.msg)}</div>
    </div>`;
  }).join('');
}
function escapeHTML(s){ return String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }

form.addEventListener('submit', (e)=>{
  e.preventDefault();
  const name = document.getElementById('wName').value.trim();
  const msg  = document.getElementById('wMsg').value.trim();
  if (!name || !msg) return;
  const list = loadWishes();
  list.push({ name, msg, t: Date.now() });
  saveWishes(list);
  form.reset();
  renderFeed();
  feed.scrollTop = 0;
});
renderFeed();

<script>
(function(){
  var KEY = 'takliflaLang';
  var q = new URLSearchParams(location.search);
  var initial = q.get('lang');
  if (!initial) { try { initial = localStorage.getItem(KEY); } catch(_){} }
  if (!initial) initial = (document.documentElement.lang || 'uz').toLowerCase().slice(0,2);
  if (['uz','en','ru'].indexOf(initial) === -1) initial = 'uz';
  document.documentElement.lang = initial;
  var sw = document.querySelector('.taklifla-lang-switch');
  if (!sw) return;
  var toggle = sw.querySelector('.taklifla-lang-toggle');
  var label = sw.querySelector('.taklifla-lang-current');
  var menu = sw.querySelector('.taklifla-lang-menu');
  function mark(active){
    if (label) label.textContent = active.toUpperCase();
    var btns = menu.querySelectorAll('button[data-lang]');
    for (var i=0;i<btns.length;i++){
      btns[i].classList.toggle('active', btns[i].getAttribute('data-lang') === active);
    }
  }
  mark(initial);
  function openMenu(open){
    sw.classList.toggle('open', open);
    if (toggle) toggle.setAttribute('aria-expanded', open ? 'true' : 'false');
  }
  toggle && toggle.addEventListener('click', function(e){
    e.stopPropagation();
    openMenu(!sw.classList.contains('open'));
  });
  menu && menu.addEventListener('click', function(e){
    var t = e.target.closest && e.target.closest('button[data-lang]');
    if (!t) return;
    var lang = t.getAttribute('data-lang');
    if (!lang) return;
    try { localStorage.setItem(KEY, lang); } catch(_){}
    document.documentElement.lang = lang;
    mark(lang);
    openMenu(false);
    window.__takliflaLang = lang;
    document.dispatchEvent(new CustomEvent('taklifla:langchange', { detail: { lang: lang } }));
  });
  document.addEventListener('click', function(e){
    if (!sw.contains(e.target)) openMenu(false);
  });
  window.__takliflaLang = initial;
})();
</script>
